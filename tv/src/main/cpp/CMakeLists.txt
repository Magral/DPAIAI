cmake_minimum_required(VERSION 3.22.1)
project(dpipilot_tv_native)
set(ENGINE_SRC engine/packets.c engine/main.c engine/conev.c engine/proxy.c engine/desync.c engine/mpool.c engine/extend.c)
add_library(byedpi SHARED ${ENGINE_SRC} tv_jni.c quic_guard.c)
target_include_directories(byedpi PRIVATE engine)
target_compile_options(byedpi PRIVATE -std=c99 -O2)
target_compile_definitions(byedpi PRIVATE _DEFAULT_SOURCE ANDROID_APP)
target_link_libraries(byedpi PRIVATE android log)
target_link_options(byedpi PRIVATE "-Wl,--wrap=sendto" "-Wl,--wrap=connect")
